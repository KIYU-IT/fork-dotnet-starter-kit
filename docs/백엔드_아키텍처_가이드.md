## 프로젝트 개요

**FullStackHero .NET 9 Starter Kit**는 엔터프라이즈급 Clean Architecture 기반의 백엔드 스타터 템플릿입니다.

### 핵심 특징

- **Modular Monolith**: 모듈식 모놀리식 아키텍처
- **Multi-Tenancy**: 내장된 멀티 테넌시 지원
- **CQRS Pattern**: MediatR 기반 명령/쿼리 분리
- **Clean Architecture**: 계층 분리 및 의존성 역전
- **API Versioning**: 버전별 API 관리
- **Minimal APIs**: Carter를 통한 경량 엔드포인트

#### FshDbContext (Spring JPA Configuration)

```csharp
// C# - Entity Framework DbContext
public abstract class FshDbContext : DbContext
{
    private readonly ICurrentUser _currentUser;

    protected FshDbContext(DbContextOptions options, ICurrentUser currentUser)
        : base(options)
    {
        _currentUser = currentUser;
    }

    // SaveChanges 오버라이드 (Auditing 자동 처리)
    public override async Task<int> SaveChangesAsync(CancellationToken cancellationToken = default)
    {
        // Audit 정보 자동 설정
        var entries = ChangeTracker.Entries<IAuditable>();
        foreach (var entry in entries)
        {
            if (entry.State == EntityState.Added)
            {
                entry.Entity.CreatedBy = Guid.Parse(_currentUser.GetUserId() ?? Guid.Empty.ToString());
                entry.Entity.CreatedOn = DateTime.UtcNow;
            }
            else if (entry.State == EntityState.Modified)
            {
                entry.Entity.LastModifiedBy = Guid.Parse(_currentUser.GetUserId() ?? Guid.Empty.ToString());
                entry.Entity.LastModifiedOn = DateTime.UtcNow;
            }
        }

        // Domain Events 발행 (Spring ApplicationEventPublisher)
        var domainEvents = ChangeTracker.Entries<BaseEntity>()
            .SelectMany(e => e.Entity.DomainEvents)
            .ToList();

        var result = await base.SaveChangesAsync(cancellationToken);

        // MediatR로 이벤트 발행
        foreach (var domainEvent in domainEvents)
        {
            await _mediator.Publish(domainEvent, cancellationToken);
        }

        return result;
    }

    // Multi-tenancy 필터 자동 적용
    protected override void OnModelCreating(ModelBuilder builder)
    {
        builder.AppendGlobalQueryFilter<IEntity>(e => e.TenantId == _currentUser.GetTenant());
        base.OnModelCreating(builder);
    }
}
```

## CQRS 패턴

### MediatR 사용 (Spring의 Service 레이어를 Command/Query로 분리)

#### Command (데이터 변경)

```csharp
// 1. Command 정의 (DTO + 의도)
public record CreateProductCommand : IRequest<CreateProductResponse>
{
    public string Name { get; init; } = default!;
    public string? Description { get; init; }
    public decimal Price { get; init; }
    public Guid BrandId { get; init; }
}

// 2. Validator (FluentValidation)
public class CreateProductValidator : AbstractValidator<CreateProductCommand>
{
    public CreateProductValidator()
    {
        RuleFor(x => x.Name)
            .NotEmpty().WithMessage("제품명은 필수입니다")
            .MaximumLength(200);

        RuleFor(x => x.Price)
            .GreaterThan(0).WithMessage("가격은 0보다 커야 합니다");

        RuleFor(x => x.BrandId)
            .NotEmpty().WithMessage("브랜드는 필수입니다");
    }
}

// 3. Handler (비즈니스 로직)
public class CreateProductHandler : IRequestHandler<CreateProductCommand, CreateProductResponse>
{
    private readonly IRepository<Product> _repository;

    public CreateProductHandler(
        [FromKeyedServices("catalog:products")] IRepository<Product> repository)
    {
        _repository = repository;
    }

    public async Task<CreateProductResponse> Handle(
        CreateProductCommand command,
        CancellationToken cancellationToken)
    {
        // 비즈니스 로직
        var product = new Product(
            command.Name,
            command.Description,
            command.Price,
            command.BrandId);

        // Domain Event 발생
        product.RegisterDomainEvent(new ProductCreated(product.Id, product.Name));

        // 저장
        await _repository.AddAsync(product, cancellationToken);

        return new CreateProductResponse(product.Id);
    }
}

// 4. Response DTO
public record CreateProductResponse(Guid Id);
```

```java
// CreateProductRequest.java (DTO)
public record CreateProductRequest(
    @NotBlank String name,
    String description,
    @Positive BigDecimal price,
    @NotNull UUID brandId
) {}

// ProductService.java
@Service
@Transactional
public class ProductService {
    private final ProductRepository productRepository;
    private final ApplicationEventPublisher eventPublisher;

    public ProductResponse createProduct(CreateProductRequest request) {
        Product product = new Product(
            request.name(),
            request.description(),
            request.price(),
            request.brandId()
        );

        product = productRepository.save(product);

        // Domain Event
        eventPublisher.publishEvent(new ProductCreated(product.getId(), product.getName()));

        return new ProductResponse(product.getId());
    }
}
```

#### Query (데이터 조회)

```csharp
// Query
public record GetProductRequest(Guid Id) : IRequest<ProductResponse>;

// Handler
public class GetProductHandler : IRequestHandler<GetProductRequest, ProductResponse>
{
    private readonly IReadRepository<Product> _repository;

    public async Task<ProductResponse> Handle(
        GetProductRequest query,
        CancellationToken cancellationToken)
    {
        // Specification Pattern (Ardalis.Specification)
        var spec = new GetProductByIdSpec(query.Id);
        var product = await _repository.FirstOrDefaultAsync(spec, cancellationToken);

        if (product is null)
            throw new ProductNotFoundException(query.Id);

        return new ProductResponse(
            product.Id,
            product.Name,
            product.Description,
            product.Price,
            product.Brand.Name
        );
    }
}

// Specification (복잡한 쿼리 캡슐화)
public class GetProductByIdSpec : Specification<Product>
{
    public GetProductByIdSpec(Guid id)
    {
        Query
            .Where(p => p.Id == id)
            .Include(p => p.Brand);  // JPA fetch join
    }
}
```

```csharp
// CreateProductEndpoint.cs
public static class CreateProductEndpoint
{
    public static RouteGroupBuilder MapProductCreationEndpoint(this RouteGroupBuilder group)
    {
        group.MapPost("/", async (
            CreateProductCommand command,
            IMediator mediator,
            CancellationToken cancellationToken) =>
        {
            var result = await mediator.Send(command, cancellationToken);
            return Results.Created($"/api/v1/catalog/products/{result.Id}", result);
        })
        .WithName("CreateProduct")
        .WithSummary("새 제품 생성")
        .WithDescription("카탈로그에 새 제품을 추가합니다")
        .Produces<CreateProductResponse>(StatusCodes.Status201Created)
        .ProducesProblem(StatusCodes.Status400BadRequest)
        .RequirePermission("Catalog", "Create");  // 권한 체크

        return group;
    }
}
```

```csharp
// TokenService.cs
public class TokenService : ITokenService
{
    public async Task<TokenResponse> GetTokenAsync(TokenGenerationCommand request)
    {
        var user = await _userManager.FindByEmailAsync(request.Email);

        // 인증 확인
        if (!await _userManager.CheckPasswordAsync(user, request.Password))
            throw new UnauthorizedException("인증 실패");

        // Claims 생성 (Spring Security Claims)
        var claims = new List<Claim>
        {
            new(ClaimTypes.NameIdentifier, user.Id.ToString()),
            new(ClaimTypes.Email, user.Email),
            new(FshClaims.Tenant, user.TenantId),
            new(FshClaims.FullName, user.FullName)
        };

        // JWT 토큰 생성
        var token = GenerateJwt(claims);
        var refreshToken = GenerateRefreshToken();

        return new TokenResponse(token, refreshToken);
    }

    private string GenerateJwt(IEnumerable<Claim> claims)
    {
        var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(_jwtOptions.Key));
        var credentials = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);

        var token = new JwtSecurityToken(
            issuer: "FSH",
            audience: "FSH",
            claims: claims,
            expires: DateTime.UtcNow.AddMinutes(_jwtOptions.TokenExpirationInMinutes),
            signingCredentials: credentials
        );

        return new JwtSecurityTokenHandler().WriteToken(token);
    }
}
```

```csharp
// CreateProductValidator.cs
public class CreateProductValidator : AbstractValidator<CreateProductCommand>
{
    private readonly IReadRepository<Brand> _brandRepository;

    public CreateProductValidator(IReadRepository<Brand> brandRepository)
    {
        _brandRepository = brandRepository;

        // 필수 검증
        RuleFor(x => x.Name)
            .NotEmpty().WithMessage("제품명은 필수입니다")
            .MaximumLength(200).WithMessage("제품명은 200자를 초과할 수 없습니다")
            .MustAsync(BeUniqueName).WithMessage("이미 존재하는 제품명입니다");

        // 숫자 검증
        RuleFor(x => x.Price)
            .GreaterThan(0).WithMessage("가격은 0보다 커야 합니다")
            .LessThanOrEqualTo(1000000).WithMessage("가격은 1,000,000을 초과할 수 없습니다");

        // 외래키 검증
        RuleFor(x => x.BrandId)
            .NotEmpty().WithMessage("브랜드는 필수입니다")
            .MustAsync(BrandMustExist).WithMessage("존재하지 않는 브랜드입니다");

        // 조건부 검증
        When(x => !string.IsNullOrEmpty(x.Description), () =>
        {
            RuleFor(x => x.Description)
                .MaximumLength(1000).WithMessage("설명은 1000자를 초과할 수 없습니다");
        });

        // 복잡한 검증
        RuleFor(x => x)
            .Must(BeValidPriceRange)
            .WithMessage("프리미엄 브랜드는 최소 가격이 10,000원 이상이어야 합니다");
    }

    // 비동기 커스텀 검증
    private async Task<bool> BeUniqueName(string name, CancellationToken cancellationToken)
    {
        var spec = new ProductByNameSpec(name);
        return !await _repository.AnyAsync(spec, cancellationToken);
    }

    private async Task<bool> BrandMustExist(Guid brandId, CancellationToken cancellationToken)
    {
        return await _brandRepository.GetByIdAsync(brandId, cancellationToken) is not null;
    }

    private bool BeValidPriceRange(CreateProductCommand command)
    {
        // 복잡한 비즈니스 룰
        if (IsPremiumBrand(command.BrandId))
        {
            return command.Price >= 10000;
        }
        return true;
    }
}
```

### 자동 검증 파이프라인

```csharp
// ValidationBehavior.cs (MediatR Pipeline)
public class ValidationBehavior<TRequest, TResponse>
    : IPipelineBehavior<TRequest, TResponse>
    where TRequest : IRequest<TResponse>
{
    private readonly IEnumerable<IValidator<TRequest>> _validators;

    public async Task<TResponse> Handle(
        TRequest request,
        RequestHandlerDelegate<TResponse> next,
        CancellationToken cancellationToken)
    {
        if (_validators.Any())
        {
            var context = new ValidationContext<TRequest>(request);

            var validationResults = await Task.WhenAll(
                _validators.Select(v => v.ValidateAsync(context, cancellationToken)));

            var failures = validationResults
                .SelectMany(r => r.Errors)
                .Where(f => f != null)
                .ToList();

            if (failures.Count != 0)
            {
                throw new ValidationException(failures);
            }
        }

        return await next();
    }
}
```
